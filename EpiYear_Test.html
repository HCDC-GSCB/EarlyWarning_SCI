df_c2 <- df_all %>%
  group_by(Disease) %>%
  mutate(
    q3_cases = quantile(Total_Cases, 0.75, na.rm = TRUE), 
    out_cases = q3_cases + 1.5 * IQR(Total_Cases, na.rm = TRUE),
    
    q3_peak = quantile(Peak_Cases, 0.75, na.rm = TRUE), 
    out_peak = q3_peak + 1.5 * IQR(Peak_Cases, na.rm = TRUE),
    
    q3_accel = quantile(Max_Accel, 0.75, na.rm = TRUE), 
    out_accel = q3_accel + 1.5 * IQR(Max_Accel, na.rm = TRUE)
  ) %>%
  ungroup() %>%
  mutate(
    Outlier_Cases = ifelse(Total_Cases > out_cases, "Outlier", "Normal"),
    Outlier_Peak = ifelse(Peak_Cases > out_peak, "Outlier", "Normal"),
    Outlier_Accel = ifelse(Max_Accel > out_accel, "Outlier", "Normal")
  )

df_c2_long <- df_c2 %>%
  select(Disease, Year, Total_Cases, Peak_Cases, Max_Accel, 
         out_cases, out_peak, out_accel, 
         Outlier_Cases, Outlier_Peak, Outlier_Accel) %>%
  pivot_longer(
    cols = c(Total_Cases, Peak_Cases, Max_Accel),
    names_to = "Metric",
    values_to = "Value"
  ) %>%
  mutate(
    Threshold = case_when(
      Metric == "Total_Cases" ~ out_cases,
      Metric == "Peak_Cases" ~ out_peak,
      Metric == "Max_Accel" ~ out_accel
    ),
    Outlier_Status = case_when(
      Metric == "Total_Cases" ~ Outlier_Cases,
      Metric == "Peak_Cases" ~ Outlier_Peak,
      Metric == "Max_Accel" ~ Outlier_Accel
    ),
    Metric = case_when(
      Metric == "Total_Cases" ~ "Total cases",
      Metric == "Peak_Cases" ~ "Peak cases",
      Metric == "Max_Accel" ~ "Max Acel"
    )
  )

df_c2_long$Metric <- factor(df_c2_long$Metric, levels = c("Total cases", "Peak cases", "Max Acel"))

run_kmeans <- function(df) {
  df_scaled <- scale(df %>% select(Total_Cases, Peak_Cases, Max_Accel))
  set.seed(42)
  km <- kmeans(df_scaled, centers = 2, nstart = 25)
  df$Cluster <- km$cluster
  mean_c <- df %>% group_by(Cluster) %>% summarise(m = mean(Total_Cases))
  epi_id <- mean_c$Cluster[which.max(mean_c$m)]
  df %>% mutate(Status = ifelse(Cluster == epi_id, "Epidemic years", "Endemic years"))
}

run_iso <- function(df) {
  model <- isolation.forest(df %>% select(Total_Cases, Peak_Cases, Max_Accel), ntrees = 100, seed = 123)
  df$Anomaly_Score <- predict(model, df %>% select(Total_Cases, Peak_Cases, Max_Accel))
  df$Iso_Status <- ifelse(df$Anomaly_Score >= 0.5, "Epidemic years", "Endemic years")
  list(data = df, model = model)
}

create_grid <- function(df, model, dis_name) {
  g_x <- seq(min(df$Total_Cases)*0.8, max(df$Total_Cases)*1.1, length.out=150)
  g_y <- seq(min(df$Peak_Cases)*0.8, max(df$Peak_Cases)*1.1, length.out=150)
  grid <- expand.grid(Total_Cases = g_x, Peak_Cases = g_y)
  grid$Max_Accel <- median(df$Max_Accel)
  grid$Grid_Score <- predict(model, grid)
  grid$Disease <- dis_name
  return(grid)
}

grid_all <- bind_rows(
